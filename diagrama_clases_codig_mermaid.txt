    namespace FactoryMethodBoletos {

            <<interface>>

        }

            -seccion: String
            +mostrarDetalles() void
        }

        class BoletoReservado {
            -numeroAsiento: String
            -fila: String
            +mostrarDetalles() void
        }

        class CreadorBoleto {
            <<abstract>>
            +procesarEmision() void
            +crearBoleto()* IBoleto
        }

        class CreadorBoletoVIP {
            +crearBoleto() IBoleto
        }

        class CreadorBoletoGeneral {
            +crearBoleto() IBoleto
        }

            +crearBoleto() IBoleto
        }
    }

    CreadorBoleto <|-- CreadorBoletoGeneral
    CreadorBoleto <|-- CreadorBoletoReservado

    namespace DecoratorPoliticas {
        class IPoliticaCompra {
            <<interface>>
            +validarCompra(comprador: Comprador, cantidad: int) bool
            +calcularReembolso(monto: float) float
            +permiteCambioFecha() bool
        }

        class PoliticaEventoBase {
            -permiteDevoluciones: bool
            -plazoDevolucionDias: int
            -porcentajeReembolso: float
            +validarCompra(comprador: Comprador, cantidad: int) bool
            +calcularReembolso(monto: float) float
            +permiteCambioFecha() bool
        }

        class PoliticaDecorator {
            <<abstract>>
            #componente: IPoliticaCompra
            +validarCompra(comprador: Comprador, cantidad: int) bool
            +calcularReembolso(monto: float) float
            +permiteCambioFecha() bool
        }

        class LimiteBoletosDecorator {
            -limiteBoletoPorUsuario: int
            +validarCompra(comprador: Comprador, cantidad: int) bool
        }

        class VerificacionEdadDecorator {
            -edadMinima: int
            +validarCompra(comprador: Comprador, cantidad: int) bool
        }

        class RestriccionSocioDecorator {
            -requiereMembresia: bool
            +validarCompra(comprador: Comprador, cantidad: int) bool
        }
    }

    IPoliticaCompra <|.. PoliticaEventoBase : implementa
    IPoliticaCompra <|.. PoliticaDecorator : implementa
    PoliticaDecorator o-- IPoliticaCompra : envuelve (componente)
    PoliticaDecorator <|-- LimiteBoletosDecorator : hereda
    PoliticaDecorator <|-- VerificacionEdadDecorator : hereda
    PoliticaDecorator <|-- RestriccionSocioDecorator : hereda

    namespace ModeloDominio {

            -id: String

            -email: String
            -telefono: String
            +getId() String
            +getEmail() String
        }

            -limiteCompra: int
            +getCanalPreferido() String
            +getLimiteCompra() int
        }

        class Organizador {
            -organizacion: String
            +getOrganizacion() String
        }

        class Evento {
            -id: String
            -nombre: String
            -artista: String
            -fecha: Date
            -lugar: String
            -genero: String
            -estado: EstadoEvento
            -capacidadTotal: int
            +getId() String
            +getEstado() EstadoEvento
            +getDisponibilidad() int
        }

        class Reserva {
            -id: String
            -fechaCreacion: Date
            -fechaExpiracion: Date
            -cantidad: int
            +estaVigente() bool
            +expirar() void
        }

        class Pago {
            -id: String
            -monto: float
            -metodo: String
            -estado: String
            -fechaPago: Date
            +getId() String
        }

        class Incidente {
            -id: String

            -tipo: String

        }

        class GestorEventos {

        }

        class GestorNotificaciones {
            -canales: ICanal[]

        }

        class GestorIncidentes {

        }
    }
