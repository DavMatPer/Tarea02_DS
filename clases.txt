    }

    class ManejadorIncidente {
        <<abstract>>
        #siguienteManejador: ManejadorIncidente
        +setSiguienteManejador(man: ManejadorIncidente) void
        +manejarIncidente(incidente: Incidente) void
    }

    class AgenteSoporte {
        -idAgente: String

    class DepartamentoAdministracion {
        -idDepartamento: String
        +manejarIncidente(incidente: Incidente) void
        +resolucionFinal(incidente: Incidente) void
    }

    ManejadorIncidente <|-- AgenteSoporte
    ManejadorIncidente <|-- DepartamentoAdministracion
